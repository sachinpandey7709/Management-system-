class UserManager:

    def __init__(self):

        self.users = {}

        self.rights_matrix = {}

        self.audit = []

        self.load_data()



    # ---------- persistence ----------

    def load_data(self):

        for file, attr in [(DATA_FILE, "users"),

                           (RIGHTS_FILE, "rights_matrix"),

                           (AUDIT_FILE, "audit")]:

            if os.path.exists(file):

                try:

                    with open(file, "r", encoding="utf-8") as f:

                        setattr(self, attr, json.load(f))

                except Exception:

                    pass



    def save_data(self):

        with open(DATA_FILE, "w", encoding="utf-8") as f:

            json.dump(self.users, f, indent=2)

        with open(RIGHTS_FILE, "w", encoding="utf-8") as f:

            json.dump(self.rights_matrix, f, indent=2)

        with open(AUDIT_FILE, "w", encoding="utf-8") as f:

            json.dump(self.audit[-1000:], f, indent=2)



    def log(self, actor, action, target=""):

        self.audit.append({

            "time": datetime.now().isoformat(timespec="seconds"),

            "actor": actor, "action": action, "target": target,

        })

        self.save_data()



    # ---------- hashing ----------

    @staticmethod

    def hash_password(password, salt=None):

        if salt is None:

            salt = secrets.token_hex(16)

        h = hashlib.sha256((salt + password).encode()).hexdigest()

        return f"{salt}${h}"



    @staticmethod

    def verify_password(password, stored):

        try:

            salt, hashed = stored.split("$")

            return hashlib.sha256((salt + password).encode()).hexdigest() == hashed

        except Exception:

            return False



    # ---------- CRUD ----------

    def create_user(self, username, password, full_name="", role="USER", is_admin=False):

        if not username or not password:

            return False, "Username and password required"

        if username in self.users:

            return False, "User already exists"

        self.users[username] = {

            "username": username,

            "password": self.hash_password(password),

            "full_name": full_name,

            "role": role,

            "is_admin": is_admin,

            "active": True,

            "created_at": datetime.now().isoformat(timespec="seconds"),

            "last_login": None,

        }

        self.rights_matrix[username] = {}

        if is_admin:

            self.grant_all_rights(username, save=False)

        self.save_data()

        return True, f"User '{username}' created"



    def update_user(self, username, **fields):

        if username not in self.users:

            return False, "User not found"

        for k, v in fields.items():

            if k == "password" and v:

                self.users[username]["password"] = self.hash_password(v)

            elif k in self.users[username] and k not in ("username", "password"):

                self.users[username][k] = v

        self.save_data()

        return True, f"User '{username}' updated"



    def delete_user(self, username):

        if username not in self.users:

            return False, "User not found"

        del self.users[username]

        self.rights_matrix.pop(username, None)

        self.save_data()

        return True, f"User '{username}' deleted"



    def authenticate(self, username, password):

        u = self.users.get(username)

        if not u:

            return False, "User not found"

        if not u["active"]:

            return False, "User is inactive"

        if not self.verify_password(password, u["password"]):

            return False, "Invalid password"

        u["last_login"] = datetime.now().isoformat(timespec="seconds")

        self.save_data()

        return True, "OK"



    # ---------- rights ----------

    def grant_all_rights(self, username, save=True):

        if username not in self.users:

            return False, "User not found"

        self.rights_matrix[username] = {

            m: {i: list(RIGHT_TYPES) for i in items}

            for m, items in DEFAULT_MODULES.items()

        }

        if save:

            self.save_data()

        return True, "All rights granted"



    def revoke_all_rights(self, username):

        if username not in self.users:

            return False, "User not found"

        self.rights_matrix[username] = {}

        self.save_data()

        return True, "All rights revoked"



    def grant_right(self, username, module, item, right):

        if username not in self.users:

            return False, "User not found"

        self.rights_matrix.setdefault(username, {}) \

            .setdefault(module, {}).setdefault(item, [])

        if right not in self.rights_matrix[username][module][item]:

            self.rights_matrix[username][module][item].append(right)

            self.save_data()

        return True, "Granted"



    def revoke_right(self, username, module, item, right):

        try:

            if right in self.rights_matrix[username][module][item]:

                self.rights_matrix[username][module][item].remove(right)

                self.save_data()

                return True, "Revoked"

        except KeyError:

            pass

        return False, "Not found"



    def has_right(self, username, module, item, right):

        if self.users.get(username, {}).get("is_admin"):

            return True

        try:

            return right in self.rights_matrix[username][module][item]

        except KeyError:

            return False



    def is_admin(self, username):

        return self.users.get(username, {}).get("is_admin", False)



    def count_rights(self, uname):

        if self.is_admin(uname):

            return sum(len(items) * len(RIGHT_TYPES)

                       for items in DEFAULT_MODULES.values())

        total = 0

        for module in self.rights_matrix.get(uname, {}).values():

            for rights in module.values():

                total += len(rights)

        return total

def export_users_csv(um, path):

    with open(path, "w", newline="", encoding="utf-8") as f:

        writer = csv.writer(f)

        writer.writerow(["username", "full_name", "role", "is_admin", "active",

                         "created_at", "last_login", "total_rights"])

        for uname, u in um.users.items():

            writer.writerow([

                uname, u["full_name"], u["role"],

                "YES" if u["is_admin"] else "NO",

                "YES" if u["active"] else "NO",

                u["created_at"], u["last_login"] or "",

                um.count_rights(uname),

            ])

    return True, f"Exported {len(um.users)} users to {os.path.basename(path)}"





def export_users_excel(um, path):

    if not HAS_EXCEL:

        return False, "openpyxl not installed.\nRun: pip install openpyxl"



    wb = Workbook()

    hfont = Font(bold=True, color="FFFFFF", size=11)

    hfill = PatternFill("solid", fgColor="7C3AED")

    center = Alignment(horizontal="center", vertical="center")



    # --- Sheet 1: Users ---

    ws = wb.active

    ws.title = "Users"

    ws.append(["Username", "Full Name", "Role", "Admin", "Active",

               "Created", "Last Login", "Total Rights"])

    for c in ws[1]:

        c.font = hfont; c.fill = hfill; c.alignment = center

    for uname, u in um.users.items():

        ws.append([uname, u["full_name"], u["role"],

                   "YES" if u["is_admin"] else "NO",

                   "YES" if u["active"] else "NO",

                   u["created_at"], u["last_login"] or "—",

                   um.count_rights(uname)])

    _autosize(ws)



    # --- Sheet 2: Rights Matrix ---

    ws2 = wb.create_sheet("Rights Matrix")

    ws2.append(["Username", "Module", "Item"] + RIGHT_TYPES)

    for c in ws2[1]:

        c.font = hfont; c.fill = hfill; c.alignment = center

    for uname in um.users:

        is_admin = um.is_admin(uname)

        for module, items in DEFAULT_MODULES.items():

            for item in items:

                row = [uname, module, item]

                for r in RIGHT_TYPES:

                    row.append("✔" if (is_admin or um.has_right(uname, module, item, r)) else "")

                ws2.append(row)

    _autosize(ws2)



    # --- Sheet 3: Audit Log ---

    ws3 = wb.create_sheet("Audit Log")

    ws3.append(["Time", "Actor", "Action", "Target"])

    for c in ws3[1]:

        c.font = hfont; c.fill = hfill; c.alignment = center

    for e in um.audit:

        ws3.append([e["time"], e["actor"], e["action"], e.get("target", "")])

    _autosize(ws3)



    wb.save(path)

    return True, f"Exported {len(um.users)} users + rights to {os.path.basename(path)}"

def _read_file(path):

    ext = path.lower().rsplit(".", 1)[-1]

    if ext == "csv":

        with open(path, "r", encoding="utf-8-sig") as f:

            return list(csv.reader(f))

    elif ext in ("xlsx", "xlsm"):

        if not HAS_EXCEL:

            return None

        wb = load_workbook(path, read_only=True, data_only=True)

        ws = wb.active

        return [[str(c) if c is not None else "" for c in row]

                for row in ws.iter_rows(values_only=True)]

    return None





def _get(row, idx):

    if idx is None or idx >= len(row):

        return ""

    val = row[idx]

    return str(val) if val is not None else ""





def _parse_bool(val, default=False):

    if not val:

        return default

    return str(val).strip().lower() in ("yes", "y", "true", "1", "✔", "x")





def _autosize(ws):

    for col in ws.columns:

        max_len = 0

        letter = get_column_letter(col[0].column)

        for cell in col:

            if cell.value:

                max_len = max(max_len, len(str(cell.value)))

        ws.column_dimensions[letter].width = min(max_len + 3, 40)





def preview_import(um, path):

    rows = _read_file(path)

    if rows is None:

        return None, "Unsupported format (use .csv or .xlsx)"

    if not rows:

        return None, "File is empty"



    headers = [h.strip().lower() for h in rows[0]]

    for req in ("username", "password"):

        if req not in headers:

            return None, f"Missing required column: '{req}'"



    idx = {h: i for i, h in enumerate(headers)}

    preview = []

    for row_num, row in enumerate(rows[1:], start=2):

        if not any(row):

            continue

        rec = {

            "row": row_num,

            "username": _get(row, idx.get("username")).strip(),

            "password": _get(row, idx.get("password")),

            "full_name": _get(row, idx.get("full_name")).strip(),

            "role": _get(row, idx.get("role")).strip() or "USER",

            "is_admin": _parse_bool(_get(row, idx.get("is_admin"))),

            "active": _parse_bool(_get(row, idx.get("active")), default=True),

        }

        if not rec["username"]:

            rec.update(status="invalid", reason="Missing username")

        elif not rec["password"]:

            rec.update(status="invalid", reason="Missing password")

        elif rec["username"] in um.users:

            rec.update(status="duplicate", reason="Username already exists")

        else:

            rec.update(status="new", reason="")

        preview.append(rec)

    return preview, "OK"





def commit_import(um, preview, actor, conflict="skip"):

    created = updated = skipped = 0

    errors = []

    for rec in preview:

        if rec["status"] == "invalid":

            skipped += 1

            errors.append(f"Row {rec['row']}: {rec['reason']}")

            continue

        if rec["status"] == "duplicate":

            if conflict == "skip":

                skipped += 1

                continue

            ok, msg = um.update_user(

                rec["username"], full_name=rec["full_name"],

                role=rec["role"], is_admin=rec["is_admin"],

                active=rec["active"])

            if ok:

                if rec["is_admin"]:

                    um.grant_all_rights(rec["username"])

                um.log(actor, "IMPORT_UPDATE", rec["username"])

                updated += 1

            else:

                errors.append(f"Row {rec['row']}: {msg}")

                skipped += 1

        else:

            ok, msg = um.create_user(rec["username"], rec["password"],

                                     rec["full_name"], rec["role"], rec["is_admin"])

            if ok:

                if not rec["active"]:

                    um.update_user(rec["username"], active=False)

                um.log(actor, "IMPORT_CREATE", rec["username"])

                created += 1

            else:

                errors.append(f"Row {rec['row']}: {msg}")

                skipped += 1

    um.save_data()

    return created, updated, skipped, errors

class LoginWindow:

    def __init__(self, root, um, on_success):

        self.root = root

        self.um = um

        self.on_success = on_success

        self.root.title("Marg Gold Edition - Login")

        self.root.geometry("420x520")

        self.root.configure(bg=COLORS["bg"])

        self.root.resizable(False, False)

        self._center()

        self._build()



    def _center(self):

        self.root.update_idletasks()

        x = (self.root.winfo_screenwidth() - 420) // 2

        y = (self.root.winfo_screenheight() - 520) // 2

        self.root.geometry(f"420x520+{x}+{y}")



    def _build(self):

        for w in self.root.winfo_children():

            w.destroy()

        outer = ttk.Frame(self.root, padding=30)

        outer.pack(fill="both", expand=True)



        ttk.Label(outer, text="💎", font=("Segoe UI", 44)).pack(pady=(10, 0))

        ttk.Label(outer, text="MARG GOLD", style="Title.TLabel").pack()

        ttk.Label(outer, text="Complete User Management",

                  style="Subtitle.TLabel").pack(pady=(0, 30))



        card = ttk.Frame(outer, style="Card.TFrame", padding=25)

        card.pack(fill="x")



        ttk.Label(card, text="Username", style="Card.TLabel").pack(anchor="w", pady=(0, 4))

        self.user_e = ttk.Entry(card, font=("Segoe UI", 11))

        self.user_e.pack(fill="x", pady=(0, 14), ipady=4)



        ttk.Label(card, text="Password", style="Card.TLabel").pack(anchor="w", pady=(0, 4))

        self.pass_e = ttk.Entry(card, show="•", font=("Segoe UI", 11))

        self.pass_e.pack(fill="x", pady=(0, 20), ipady=4)



        ttk.Button(card, text="LOGIN", style="Accent.TButton",

                   command=self._login).pack(fill="x", ipady=4)

        ttk.Button(card, text="Exit", command=self.root.destroy) \

            .pack(fill="x", pady=(8, 0), ipady=2)



        self.status = ttk.Label(outer, text="", style="Subtitle.TLabel")

        self.status.pack(pady=(15, 0))



        self.user_e.focus()

        self.root.bind("<Return>", lambda e: self._login())



    def _login(self):

        u = self.user_e.get().strip()

        p = self.pass_e.get()

        ok, msg = self.um.authenticate(u, p)

        if ok:

            self.um.log(u, "LOGIN")

            self.on_success(u)

        else:

            self.status.configure(text=f"❌ {msg}")

            self.pass_e.delete(0, "end")

    def refresh_user_tree(self):

        if not hasattr(self, "user_tree"):

        for row in self.user_tree.get_children():

            self.user_tree.delete(row)

        q = self.search_var.get().lower() if hasattr(self, "search_var") else ""

        for uname, u in self.um.users.items():

            if q and q not in uname.lower() and q not in u["full_name"].lower():

                uname, u["full_name"], u["role"],

                "✔" if u["is_admin"] else "",

                (u["last_login"] or "—").replace("T", " "),

    def refresh_rights(self):

        if not hasattr(self, "rights_tree"):

            return

        for row in self.rights_tree.get_children():

            self.rights_tree.delete(row)

        uname = self.rights_user.get()

        is_admin = self.um.is_admin(uname)

        for module, items in DEFAULT_MODULES.items():

            for item in items:

                vals = [module, item]

                for r in RIGHT_TYPES:

                    if is_admin:

                        vals.append("✔")

                    else:

                        vals.append("✔" if self.um.has_right(uname, module, item, r) else "")

                self.rights_tree.insert("", "end", values=vals,

                                        tags=(f"{module}|{item}",))



    def _toggle_right_click(self, event):

        uname = self.rights_user.get()

        if self.um.is_admin(uname):

            messagebox.showinfo("Admin", "Admin has all rights automatically")

            return

        row = self.rights_tree.identify_row(event.y)

        col = self.rights_tree.identify_column(event.x)

        if not row or not col:

            return

        module, item = self.rights_tree.item(row, "tags")[0].split("|")

        ci = int(col.replace("#", "")) - 1

        if ci < 2:

            return

        right = RIGHT_TYPES[ci - 2]

        if self.um.has_right(uname, module, item, right):

            self.um.revoke_right(uname, module, item, right)

        else:

            self.um.grant_right(uname, module, item, right)

        self.um.log(self.current_user, "TOGGLE_RIGHT",

                    f"{uname}:{module}/{item}/{right}")

        self.refresh_rights()



    def grant_all(self):

        uname = self.rights_user.get()

        self.um.grant_all_rights(uname)

        self.um.log(self.current_user, "GRANT_ALL", uname)

        self.refresh_rights()



    def revoke_all(self):

        uname = self.rights_user.get()

        if uname == self.current_user:

            messagebox.showerror("Error", "Cannot revoke your own rights")

            return

        self.um.revoke_all_rights(uname)

        self.um.log(self.current_user, "REVOKE_ALL", uname)

        self.refresh_rights()

                messagebox.showerror("Error", "Passwords don't match", parent=dlg)

                return

            if len(n1.get()) < 3:

                messagebox.showerror("Error", "Password too short", parent=dlg)

                return

            self.um.update_user(self.current_user, password=n1.get())

            self.um.log(self.current_user, "CHANGE_PASSWORD")

            messagebox.showinfo("Success", "Password changed", parent=dlg)

            dlg.destroy()



        ttk.Button(dlg, text="Save", style="Accent.TButton",

                   command=save).pack(pady=15)



    def logout(self):

        if messagebox.askyesno("Logout", "Are you sure?"):

            self.um.log(self.current_user, "LOGOUT")

            self.root.destroy()

            launch()

class UserDialog:

    def __init__(self, parent, um, actor, edit_user=None, on_save=None):

        self.um = um

        self.actor = actor

        self.edit_user = edit_user

        self.on_save = on_save

        self.dlg = tk.Toplevel(parent)

        self.dlg.title("Edit User" if edit_user else "Create User")

        self.dlg.configure(bg=COLORS["bg"])

        self.dlg.geometry("460x520")

        self.dlg.transient(parent); self.dlg.grab_set()

        self._build()



    def _build(self):

        ttk.Label(self.dlg, text="Edit User" if self.edit_user else "New User",

                  font=("Segoe UI", 16, "bold")).pack(pady=(20, 15))

        form = ttk.Frame(self.dlg, padding=25)

        form.pack(fill="both", expand=True)



        def add(label, show=None):

            ttk.Label(form, text=label).pack(anchor="w", pady=(8, 2))

            e = ttk.Entry(form, show=show or ""); e.pack(fill="x", ipady=4)

            return e



        self.username_e = add("Username")

        self.fullname_e = add("Full Name")

        self.password_e = add("Password", show="•")

        self.role_e = add("Role")



        self.admin_var = tk.BooleanVar()

        ttk.Checkbutton(form, text="Administrator (all rights automatically)",

                        variable=self.admin_var).pack(anchor="w", pady=10)

        self.active_var = tk.BooleanVar(value=True)

        ttk.Checkbutton(form, text="Account Active",

                        variable=self.active_var).pack(anchor="w")



        if self.edit_user:

            u = self.um.users[self.edit_user]

            self.username_e.insert(0, u["username"])

            self.username_e.configure(state="disabled")

            self.fullname_e.insert(0, u["full_name"])

            self.role_e.insert(0, u["role"])

            self.admin_var.set(u["is_admin"])

            self.active_var.set(u["active"])

            self.password_e.configure(state="disabled")

            ttk.Label(form, text="(password unchanged — use Reset Password)",

                      foreground=COLORS["text_dim"]).pack(anchor="w")



        btns = ttk.Frame(form); btns.pack(fill="x", pady=20)

        ttk.Button(btns, text="Cancel",

                   command=self.dlg.destroy).pack(side="right", padx=5)

        ttk.Button(btns, text="💾 Save", style="Accent.TButton",

                   command=self._save).pack(side="right", padx=5)



    def _save(self):

        uname = self.username_e.get().strip()

        fullname = self.fullname_e.get().strip()

        role = self.role_e.get().strip() or "USER"

        is_admin = self.admin_var.get()

        active = self.active_var.get()



        if self.edit_user:

            ok, msg = self.um.update_user(

                self.edit_user, full_name=fullname, role=role,

                is_admin=is_admin, active=active)

            if ok and is_admin:

                self.um.grant_all_rights(self.edit_user)

            self.um.log(self.actor, "UPDATE_USER", self.edit_user)

        else:

            pw = self.password_e.get()

            ok, msg = self.um.create_user(uname, pw, fullname, role, is_admin)

            if ok:

                self.um.log(self.actor, "CREATE_USER", uname)



        if ok:

            messagebox.showinfo("Success", msg)

            if self.on_save:

                self.on_save()

            self.dlg.destroy()

        else:

            messagebox.showerror("Error", msg)

def launch():

    um = UserManager()



    # Bootstrap first admin

    if not um.users:

        boot = tk.Tk(); boot.withdraw()

        messagebox.showinfo("First Run",

                            "No users exist.\n\nPlease create the ADMIN account.")

        admin_u = simpledialog.askstring("Admin Username", "Enter admin username:")

        admin_p = simpledialog.askstring("Admin Password", "Enter admin password:",

                                          show="•")

        if admin_u and admin_p:

            um.create_user(admin_u, admin_p, "System Administrator",

                           role="ADMIN", is_admin=True)

            um.log(admin_u, "BOOTSTRAP_ADMIN")

        boot.destroy()



    root = tk.Tk()

    setup_styles(root)



    def on_login(username):

        root.destroy()

        r = tk.Tk()

        setup_styles(r)

        MargApp(r, um, username)

        r.mainloop()



    LoginWindow(root, um, on_login)

    root.mainloop()





if __name__ == "__main__":

    launch()
